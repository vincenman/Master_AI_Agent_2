```python
# Python program to calculate the first 10,000 terms of the series:
# 1 - 1/3 + 1/5 - 1/7 + ... and multiply the total by 4

def calculate_series(num_terms):
    total = 0.0
    for i in range(num_terms):
        # Calculate the denominator (odd numbers: 1, 3, 5, 7, ...)
        denominator = 2 * i + 1
        # Add or subtract term depending on whether i is even or odd
        term = (-1) ** i / denominator
        total += term
    return total * 4

if __name__ == "__main__":
    terms = 10000
    result = calculate_series(terms)
    print(f"Approximation of pi using {terms} terms: {result}")
```
